        for(let i=0; i<40; i++) {
            hexParticles.push({
                x: Math.random()*canvas.width, y: Math.random()*canvas.height,
                size: Math.random()*25+10, speedY: Math.random()*0.5+0.2, opacity: Math.random()*0.4+0.1
            });

        function drawHexagon(x, y, r) {
            ctx.beginPath();
            for (let i = 0; i < 6; i++) { ctx.lineTo(x + r * Math.cos(i * Math.PI / 3), y + r * Math.sin(i * Math.PI / 3)); }
            ctx.closePath();
        }
        function animate() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            hexParticles.forEach(p => {
                p.y -= p.speedY; if(p.y < -50) p.y = canvas.height + 50;
                ctx.strokeStyle = `rgba(255, 0, 60, ${p.opacity})`; ctx.lineWidth = 1.5;
                drawHexagon(p.x, p.y, p.size); ctx.stroke();
            });
